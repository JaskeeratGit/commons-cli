@ExtendWith(MockitoExtension.class)
public class Options_hasOption_14_0_Test {

    private Options options;

    @BeforeEach
    public void setUp() {
        options = new Options();
    }

    @Test
    public void testHasShortOptionExactAndWithHyphen() throws Exception {
        putOptionIntoField("shortOpts", "a");
        // exact
        assertTrue(options.hasOption("a"));
        // single hyphen
        assertTrue(options.hasOption("-a"));
        // double hyphen also strips to "a"
        assertTrue(options.hasOption("--a"));
    }

    @Test
    public void testHasLongOptionExactAndWithHyphens() throws Exception {
        putOptionIntoField("longOpts", "help");
        assertTrue(options.hasOption("help"));
        // single hyphen leads to "help"
        assertTrue(options.hasOption("-help"));
        // double hyphen leads to "help"
        assertTrue(options.hasOption("--help"));
    }

    @Test
    public void testAbsentOption() throws Exception {
        putOptionIntoField("shortOpts", "x");
        putOptionIntoField("longOpts", "longX");
        assertFalse(options.hasOption("y"));
        assertFalse(options.hasOption("-y"));
        assertFalse(options.hasOption("--y"));
        // ensure unrelated formats do not match stored keys
        assertFalse(options.hasOption("---y"));
    }

    @Test
    public void testBothShortAndLongPresent() throws Exception {
        putOptionIntoField("shortOpts", "dup");
        putOptionIntoField("longOpts", "dup");
        // either source should result in true
        assertTrue(options.hasOption("dup"));
        assertTrue(options.hasOption("-dup"));
        assertTrue(options.hasOption("--dup"));
    }

    @Test
    public void testWeirdHyphensDoNotMatchShortKey() throws Exception {
        putOptionIntoField("shortOpts", "a");
        // "---a" is stripped of only two leading hyphens, resulting in "-a", which should not match "a"
        assertFalse(options.hasOption("---a"));
    }

    @Test
    public void testEmptyStringKeyAndHyphenVariants() throws Exception {
        // put an empty key into short options and verify behavior with hyphen variants
        putOptionIntoField("shortOpts", "");
        // exact empty string
        assertTrue(options.hasOption(""));
        // single "-" strips to empty string
        assertTrue(options.hasOption("-"));
        // double "--" strips to empty string
        assertTrue(options.hasOption("--"));
    }

    // Helper to inject Option instances into private Maps shortOpts/longOpts via reflection
    @SuppressWarnings("unchecked")
    private void putOptionIntoField(String fieldName, String key) throws Exception {
        Field field = Options.class.getDeclaredField(fieldName);
        field.setAccessible(true);
        Map<String, Option> map = (Map<String, Option>) field.get(options);
        // create a mock of the real org.apache.commons.cli.Option class to avoid constructor issues
        Option optMock = mock(Option.class);
        map.put(key, optMock);
    }
}
