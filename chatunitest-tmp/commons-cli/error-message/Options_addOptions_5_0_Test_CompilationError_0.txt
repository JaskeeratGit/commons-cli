class Options_addOptions_5_0_Test {

    @Test
    void testAddOptionsSuccessfullyMergesOptionsAndRequiredOptions() throws Exception {
        Options base = new Options();
        Options incoming = new Options();
        // base has option "a"
        Option a = new Option("a", false, "option a");
        base.addOption(a);
        // incoming has option "b" and it is required
        Option b = new Option("b", false, "option b");
        b.setRequired(true);
        incoming.addOption(b);
        // incoming also has a long option "longC" (option key "c")
        Option c = new Option("c", "longC", true, "option c");
        incoming.addOption(c);
        // perform merge
        Options returned = base.addOptions(incoming);
        // chaining: should return same base instance
        assertSame(base, returned);
        // public API checks
        assertTrue(base.hasOption("a"), "base should still have option a");
        assertTrue(base.hasOption("b"), "base should now have option b from incoming");
        assertTrue(base.hasOption("c"), "base should now have option c from incoming");
        // hasOption should also recognize long option name
        assertTrue(base.hasOption("longC"), "base should recognize long option longC");
        // required options list should include "b" key
        List<?> required = base.getRequiredOptions();
        assertNotNull(required);
        assertTrue(required.contains(b.getKey()), "required options should contain key for b");
        // getOptions() should contain options with keys a, b, c
        Collection<Option> opts = base.getOptions();
        assertNotNull(opts);
        assertEquals(3, opts.size(), "there should be three options after merge");
        assertTrue(opts.stream().map(Option::getKey).collect(Collectors.toSet()).containsAll(java.util.Set.of("a", "b", "c")));
        // Use reflection to inspect internal private shortOpts map to ensure internal state
        Field shortOptsField = Options.class.getDeclaredField("shortOpts");
        shortOptsField.setAccessible(true);
        @SuppressWarnings("unchecked")
        Map<String, Option> shortOptsMap = (Map<String, Option>) shortOptsField.get(base);
        assertTrue(shortOptsMap.containsKey("a"));
        assertTrue(shortOptsMap.containsKey("b"));
        assertTrue(shortOptsMap.containsKey("c"));
    }

    @Test
    void testAddOptionsThrowsOnDuplicateKey() {
        Options base = new Options();
        Options incoming = new Options();
        // both have option "dup"
        Option first = new Option("dup", false, "first");
        base.addOption(first);
        Option dup = new Option("dup", false, "duplicate");
        incoming.addOption(dup);
        IllegalArgumentException ex = assertThrows(IllegalArgumentException.class, () -> base.addOptions(incoming));
        assertEquals("Duplicate key: " + dup.getKey(), ex.getMessage());
    }

    @Test
    void testAddOptionsIteratesOptionGroupsEvenWhenEmpty() {
        // This test ensures the optionGroups.forEach(...) line is executed without error
        Options base = new Options();
        Options incoming = new Options();
        // incoming has no option groups; just ensure addOptions runs normally
        Option x = new Option("x", false, "x");
        incoming.addOption(x);
        // Should not throw
        Options returned = base.addOptions(incoming);
        assertSame(base, returned);
        assertTrue(base.hasOption("x"));
        // Also assert getOptionGroups on incoming returns a collection (possibly empty)
        Collection<OptionGroup> groups = incoming.getOptionGroups();
        assertNotNull(groups);
    }
}
