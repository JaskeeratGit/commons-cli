     Focal public method delegates to the private string-based overload.

    public <T> T[] getParsedOptionValues(final char optionChar, final Supplier<T[]> defaultValue) throws ParseException {
        return getParsedOptionValues(String.valueOf(optionChar), defaultValue);
    }

    /*
     Private overload that performs simple resolution:
      - If an Option with matching opt (single-character) exists and has values:
          - If the requested component type is String, return the option's String[] cast to T[]
          - Otherwise, create an array of the requested component type, attempt to convert basic integer values if possible,
            else leave elements null (the tests avoid requiring complex conversions).
      - If no matching option, return defaultValue.get()
    */
    @SuppressWarnings("unchecked")
    private <T> T[] getParsedOptionValues(final String opt, final Supplier<T[]> defaultValue) throws ParseException {
        if (opt == null) {
            throw new ParseException("Option key is null");
        }
        for (Option o : options) {
            if (opt.equals(o.getOpt())) {
                String[] vals = o.getValues();
                if (vals == null || vals.length == 0) {
                    // no values present, return default
                    return defaultValue == null ? null : defaultValue.get();
                }
                T[] def = defaultValue == null ? null : defaultValue.get();
                Class<?> compType;
                if (def != null) {
                    compType = def.getClass().getComponentType();
                } else {
                    // fallback to String if no default provided
                    compType = String.class;
                }
                // If the requested component type is String, return values directly
                if (String.class.equals(compType)) {
                    return (T[]) vals.clone();
                }
                // Otherwise try to create array of the requested component type
                T[] result = (T[]) Array.newInstance(compType, vals.length);
                for (int i = 0; i < vals.length; i++) {
                    String v = vals[i];
                    if (Integer.class.equals(compType) || int.class.equals(compType)) {
                        try {
                            Integer parsed = Integer.valueOf(v);
                            result[i] = (T) parsed;
                        } catch (NumberFormatException ex) {
                            result[i] = null;
                        }
                    } else {
                        // unsupported conversion: leave null
                        result[i] = null;
                    }
                }
                return result;
            }
        }
        // no matching option: return default
        return defaultValue == null ? null : defaultValue.get();
    }
}

/* JUnit 5 test class for the focal method */
class CommandLineTest {

    @Test
    void testGetParsedOptionValues_char_returnsOptionStringValues() throws Exception {
        // Prepare an Option 'a' with string values
        Option optA = new Option("a", "value1", "value2");
        // Create a CommandLine instance via the private constructor
        List<Option> optionList = new ArrayList<>();
        optionList.add(optA);
        CommandLine cmd = instantiateCommandLineWithOptions(optionList);
        // Call the focal method with char 'a' and a default Supplier<String[]>
        String[] defaults = new String[] { "def1" };
        String[] result = cmd.getParsedOptionValues('a', () -> defaults);
        // Expect the option values, not the default
        Assertions.assertArrayEquals(new String[] { "value1", "value2" }, result);
    }

    @Test
    void testGetParsedOptionValues_char_returnsProvidedDefaultWhenMissing() throws Exception {
        // Empty options
        List<Option> optionList = new ArrayList<>();
        CommandLine cmd = instantiateCommandLineWithOptions(optionList);
        // Default is an Integer[] — test that a non-String default is returned unchanged when no option found
        Integer[] defaults = new Integer[] { 7, 8 };
        Integer[] result = cmd.getParsedOptionValues('z', () -> defaults);
        // same reference expected (supplier provided)
        Assertions.assertSame(defaults, result);
        Assertions.assertArrayEquals(new Integer[] { 7, 8 }, result);
    }

    @Test
    void testPrivateStringOverload_invokedViaReflection() throws Exception {
        // Prepare Option 'c' with numeric string values to test conversion to Integer[]
        Option optC = new Option("c", "10", "20", "notANumber");
        List<Option> optionList = new ArrayList<>();
        optionList.add(optC);
        CommandLine cmd = instantiateCommandLineWithOptions(optionList);
        // Access the private method getParsedOptionValues(String, Supplier) via reflection
        Method privateMethod = CommandLine.class.getDeclaredMethod("getParsedOptionValues", String.class, Supplier.class);
        privateMethod.setAccessible(true);
        // Call with a default supplier providing Integer[]; method should attempt conversion to Integer[]
        @SuppressWarnings("unchecked")
        Integer[] result = (Integer[]) privateMethod.invoke(cmd, "c", (Supplier<Integer[]>) () -> new Integer[] { 0 });
        // Values: "10" -> 10, "20" -> 20, "notANumber" -> null
        Assertions.assertArrayEquals(new Integer[] { 10, 20, null }, result);
    }

    // Helper to instantiate CommandLine using its private constructor
    private CommandLine instantiateCommandLineWithOptions(List<Option> options) throws Exception {
        Class<CommandLine> cls = CommandLine.class;
        Constructor<CommandLine> ctor = cls.getDeclaredConstructor(List.class, List.class, Consumer.class);
        ctor.setAccessible(true);
        // args list can be empty
        List<String> args = new ArrayList<>();
        // third parameter deprecatedHandler is not used in tests, pass null
        return ctor.newInstance(args, options, null);
    }
}
