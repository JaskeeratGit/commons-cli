    /**
     * Create a CommandLine instance by reflectively invoking the non-public constructor:
     * CommandLine(List<String> args, List<Option> options, Consumer<Option> deprecatedHandler)
     */
    private CommandLine createCommandLine() throws Exception {
        Constructor<CommandLine> ctor = CommandLine.class.getDeclaredConstructor(List.class, List.class, Consumer.class);
        ctor.setAccessible(true);
        // Provide empty lists and a null deprecatedHandler to avoid depending on Builder
        return ctor.newInstance(new LinkedList<String>(), new ArrayList<>(), null);
    }

    @Test
    public void returnsProvidedDefaultString() throws Exception {
        CommandLine cmd = createCommandLine();
        OptionGroup group = new OptionGroup();
        String defaultValue = "default";
        String result = cmd.getParsedOptionValue(group, defaultValue);
        // Should return the exact default value supplied
        assertSame(defaultValue, result);
    }

    @Test
    public void returnsProvidedDefaultInteger() throws Exception {
        CommandLine cmd = createCommandLine();
        OptionGroup group = new OptionGroup();
        Integer defaultValue = Integer.valueOf(42);
        Integer result = cmd.getParsedOptionValue(group, defaultValue);
        assertSame(defaultValue, result);
        assertEquals(42, result.intValue());
    }

    @Test
    public void returnsProvidedDefaultNull() throws Exception {
        CommandLine cmd = createCommandLine();
        OptionGroup group = new OptionGroup();
        // Passing a null default should result in a null return (no NPE)
        String result = cmd.getParsedOptionValue(group, (String) null);
        assertNull(result);
    }

    @Test
    public void worksWhenOptionGroupIsNull() throws Exception {
        CommandLine cmd = createCommandLine();

        String result = cmd.getParsedOptionValue(null, defaultValue);

        assertSame(defaultValue, result);

    public void supportsArrayDefaultValues() throws Exception {

        String[] defaultArray = new String[] { "a", "b" };
        String[] result = cmd.getParsedOptionValue(group, defaultArray);
        // Ensure arrays are returned as-is (same reference)
        assertSame(defaultArray, result);
