    /**
     * Strips leading hyphens from the option name.
     * Returns null if input is null or the result is empty.
     */
    public static String stripLeadingHyphens(final String option) {
        if (option == null) {
            return null;
        }
        int i = 0;
        while (i < option.length() && option.charAt(i) == '-') {
